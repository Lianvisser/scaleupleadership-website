(function(){
  var API = '/.netlify/functions/bestel-boek';
  // Reservewaarden. De echte bedragen staan in netlify/lib/boek.mjs
  var cfg = { prijs: 24.95, verzending: 3.95, gratisVerzendingVanaf: 3, maxAantal: 10 };
  var $ = function(id){ return document.getElementById(id); };
  var fmt = function(v){ return '€ ' + v.toFixed(2).replace('.', ','); };
  var dlg = $('sul-bestel'), form = $('sul-form'), aantal = $('sul-aantal');

  function reken(){
    var n = Math.max(1, Math.min(cfg.maxAantal, parseInt(aantal.value, 10) || 1));
    var verz = (cfg.gratisVerzendingVanaf > 0 && n >= cfg.gratisVerzendingVanaf) ? 0 : cfg.verzending;
    var sub = Math.round(n * cfg.prijs * 100) / 100;
    $('sul-t-regel').textContent = n + 'x boek';
    $('sul-t-sub').textContent = fmt(sub);
    $('sul-t-verz').textContent = verz ? fmt(verz) : 'gratis';
    $('sul-t-tot').textContent = fmt(Math.round((sub + verz) * 100) / 100);
    if ($('bk-prijs')) $('bk-prijs').textContent = fmt(cfg.prijs);
    if ($('bk-verz')) $('bk-verz').textContent = 'inclusief btw, verzending ' + fmt(cfg.verzending);
    return n;
  }
  fetch(API).then(function(r){ return r.ok ? r.json() : null; }).then(function(d){ if (d) { cfg = d; aantal.max = d.maxAantal; reken(); } }).catch(function(){});
  reken();

  function open(e){ if (e) e.preventDefault(); if (dlg.showModal) dlg.showModal(); else dlg.setAttribute('open', ''); }
  function dicht(){ if (dlg.close) dlg.close(); else dlg.removeAttribute('open'); }

  // Elke bestaande knop naar Managementboek opent nu het formulier
  document.querySelectorAll('a[href*="managementboek"], a[href="#bestellen"]').forEach(function(a){
    a.addEventListener('click', open);
    a.setAttribute('href', '#bestellen');
    a.removeAttribute('target');
  });
  if (location.hash === '#bestellen') open();

  $('sul-sluit').addEventListener('click', dicht);
  dlg.addEventListener('click', function(e){ if (e.target === dlg) dicht(); });
  document.querySelectorAll('#sul-bestel [data-stap]').forEach(function(b){
    b.addEventListener('click', function(){
      aantal.value = Math.max(1, Math.min(cfg.maxAantal, (parseInt(aantal.value, 10) || 1) + parseInt(b.dataset.stap, 10)));
      reken();
    });
  });
  aantal.addEventListener('input', reken);

  function fout(t){ var f = $('sul-fout'); f.textContent = t; f.style.display = t ? 'block' : 'none'; }

  form.addEventListener('submit', function(e){
    e.preventDefault(); fout('');
    var verplicht = ['naam','email','straat','postcode','plaats'];
    for (var i = 0; i < verplicht.length; i++) {
      if (!form[verplicht[i]].value.trim()) { form[verplicht[i]].focus(); return fout('Vul alle velden met een * in.'); }
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.value.trim())) { form.email.focus(); return fout('Dit e-mailadres klopt niet.'); }
    if (!$('sul-akkoord').checked) return fout('Ga akkoord met de algemene voorwaarden.');

    var knop = $('sul-betaal');
    knop.disabled = true; knop.firstChild.nodeValue = 'Even geduld ';
    fetch(API, {
      method: 'POST', headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({
        aantal: reken(), naam: form.naam.value, email: form.email.value, bedrijf: form.bedrijf.value,
        straat: form.straat.value, postcode: form.postcode.value, plaats: form.plaats.value, land: form.land.value,
        opmerking: form.opmerking.value, website: form.website.value, akkoord: true
      })
    }).then(function(r){ return r.json(); }).then(function(d){
      if (d.checkoutUrl) { location.href = d.checkoutUrl; return; }
      throw new Error(d.fout || 'Er ging iets mis.');
    }).catch(function(err){
      fout(err.message || 'Er ging iets mis. Probeer het opnieuw.');
      knop.disabled = false; knop.firstChild.nodeValue = 'Afrekenen ';
    });
  });
})();
</script>
